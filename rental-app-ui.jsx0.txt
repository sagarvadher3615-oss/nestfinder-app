        <div style={{ position: "absolute", bottom: 0, left: 0, right: 0, padding: "16px 20px 32px", background: "white", borderTop: `1px solid ${theme.gray100}`, display: "flex", gap: 12 }}>
          <button style={{ flex: 1, background: theme.lightBlue, border: "none", borderRadius: 14, padding: "14px", color: theme.blue, fontSize: 14, fontWeight: 700, cursor: "pointer" }}>Schedule Visit</button>
          <button onClick={() => setShowBooking(true)} style={{ flex: 2, background: l.available ? theme.blue : theme.gray300, border: "none", borderRadius: 14, padding: "14px", color: "white", fontSize: 14, fontWeight: 700, cursor: l.available ? "pointer" : "default" }}>{l.available ? "Book Now" : "Not Available"}</button>
        </div>
        {showBooking && (
          <div style={{ position: "absolute", inset: 0, background: "rgba(0,0,0,0.5)", display: "flex", alignItems: "flex-end", borderRadius: 40 }}>
            <div style={{ background: "white", borderRadius: "24px 24px 0 0", padding: "24px 20px 40px", width: "100%" }}>
              <div style={{ fontSize: 18, fontWeight: 700, color: theme.navy, marginBottom: 4 }}>Confirm Booking</div>
              <div style={{ fontSize: 13, color: theme.gray500, marginBottom: 20 }}>{l.title}</div>
              {[{ label: "Monthly Rent", value: `₹${l.rent.toLocaleString()}` }, { label: "Security Deposit", value: `₹${l.deposit.toLocaleString()}` }, { label: "First Payment", value: `₹${(l.rent + l.deposit).toLocaleString()}` }].map((row, i) => (
                <div key={i} style={{ display: "flex", justifyContent: "space-between", padding: "10px 0", borderBottom: i < 2 ? `1px solid ${theme.gray100}` : "none" }}>
                  <span style={{ fontSize: 13, color: theme.gray500 }}>{row.label}</span>
                  <span style={{ fontSize: 13, fontWeight: 700, color: i === 2 ? theme.blue : theme.navy }}>{row.value}</span>
                </div>
              ))}
              <div style={{ display: "flex", gap: 12, marginTop: 20 }}>
                <button onClick={() => setShowBooking(false)} style={{ flex: 1, background: theme.gray100, border: "none", borderRadius: 14, padding: 14, fontSize: 14, fontWeight: 600, cursor: "pointer", color: theme.gray700 }}>Cancel</button>
                <button style={{ flex: 2, background: theme.blue, border: "none", borderRadius: 14, padding: 14, fontSize: 14, fontWeight: 700, color: "white", cursor: "pointer" }}>Pay & Book →</button>
              </div>
            </div>
          </div>
        )}
      </div>
    );
  }

  // ─── MAIN APP ─────────────────────────────────────────────────────────────
  return (
    <div style={{ width: 390, height: 844, background: theme.gray50, display: "flex", flexDirection: "column", fontFamily: "sans-serif", position: "relative", overflow: "hidden", borderRadius: 40, boxShadow: "0 40px 80px rgba(0,0,0,0.25)" }}>

      {activeTab === "home" && (
        <div style={{ flex: 1, overflowY: "auto", paddingBottom: 80 }}>
          <div style={{ background: `linear-gradient(135deg, ${theme.navy} 0%, #1A3A6B 100%)`, padding: "52px 20px 28px", borderRadius: "0 0 28px 28px" }}>
            <div style={{ display: "flex", justifyContent: "space-between", alignItems: "flex-start", marginBottom: 20 }}>
              <div>
                <div style={{ color: "rgba(255,255,255,0.6)", fontSize: 13 }}>Good morning 👋</div>
                <div style={{ color: "white", fontSize: 22, fontWeight: 700, marginTop: 2 }}>Find Your Perfect Home</div>
              </div>
              <div style={{ width: 42, height: 42, borderRadius: "50%", background: "rgba(255,255,255,0.15)", display: "flex", alignItems: "center", justifyContent: "center", fontSize: 20 }}>👤</div>
            </div>
            <div style={{ background: "white", borderRadius: 14, padding: "12px 16px", display: "flex", alignItems: "center", gap: 10 }}>
              <span style={{ fontSize: 18 }}>🔍</span>
              <input placeholder="Search area, city..." style={{ border: "none", outline: "none", flex: 1, fontSize: 14, color: theme.gray700, background: "transparent" }} />
              <div style={{ background: theme.blue, borderRadius: 10, padding: "6px 12px", color: "white", fontSize: 12, fontWeight: 600, cursor: "pointer" }}>Search</div>
            </div>
          </div>

          <div style={{ padding: "20px 20px 0" }}>
            <div style={{ display: "flex", gap: 12, marginBottom: 24 }}>
              {[{ icon: "🏠", count: "2,400+", label: "Listings" }, { icon: "🏙️", count: "12", label: "Cities" }, { icon: "✓", count: "1,800+", label: "Verified" }].map((s, i) => (
                <div key={i} style={{ flex: 1, background: "white", borderRadius: 14, padding: "14px 10px", textAlign: "center", boxShadow: "0 2px 8px rgba(0,0,0,0.06)" }}>
                  <div style={{ fontSize: 22, marginBottom: 2 }}>{s.icon}</div>
                  <div style={{ fontSize: 15, fontWeight: 800, color: theme.navy }}>{s.count}</div>
                  <div style={{ fontSize: 10, color: theme.gray500 }}>{s.label}</div>
                </div>
              ))}
            </div>

            <div style={{ marginBottom: 16 }}>
              <div style={{ fontSize: 16, fontWeight: 700, color: theme.navy, marginBottom: 12 }}>Property Type</div>
              <div style={{ display: "flex", gap: 8, overflowX: "auto", paddingBottom: 4 }}>
                {filters.map(f => <button key={f} onClick={() => setActiveFilter(f)} style={{ background: activeFilter === f ? theme.blue : "white", color: activeFilter === f ? "white" : theme.gray700, border: `1px solid ${activeFilter === f ? theme.blue : theme.gray300}`, borderRadius: 20, padding: "7px 16px", fontSize: 12, fontWeight: 600, cursor: "pointer", whiteSpace: "nowrap" }}>{f}</button>)}
              </div>
            </div>

            <div style={{ fontSize: 16, fontWeight: 700, color: theme.navy, marginBottom: 12 }}>{filteredListings.length} Properties Found</div>
            <div style={{ display: "flex", flexDirection: "column", gap: 16 }}>
              {filteredListings.map(l => (
                <div key={l.id} onClick={() => { setSelectedListing(l); setScreen("detail"); }} style={{ background: "white", borderRadius: 20, boxShadow: "0 2px 12px rgba(0,0,0,0.07)", overflow: "hidden", cursor: "pointer" }}>
                  <div style={{ height: 160, background: `linear-gradient(135deg, ${theme.navy}22, ${theme.blue}33)`, display: "flex", alignItems: "center", justifyContent: "center", position: "relative" }}>
                    <div style={{ fontSize: 70 }}>{l.img}</div>
                    <div style={{ position: "absolute", top: 12, left: 12, background: typeColors[l.type] || theme.blue, color: "white", padding: "3px 10px", borderRadius: 20, fontSize: 11, fontWeight: 600 }}>{l.type}</div>
                    <button onClick={(e) => { e.stopPropagation(); toggleSave(l.id); }} style={{ position: "absolute", top: 10, right: 12, background: "white", border: "none", borderRadius: "50%", width: 32, height: 32, cursor: "pointer", fontSize: 14, boxShadow: "0 2px 8px rgba(0,0,0,0.1)" }}>{savedIds.includes(l.id) ? "❤️" : "🤍"}</button>
                    {!l.available && <div style={{ position: "absolute", bottom: 0, left: 0, right: 0, background: "rgba(0,0,0,0.5)", color: "white", textAlign: "center", fontSize: 12, padding: "6px", fontWeight: 600 }}>Currently Occupied</div>}
                  </div>
                  <div style={{ padding: "14px 16px" }}>
                    <div style={{ display: "flex", justifyContent: "space-between", alignItems: "flex-start" }}>
                      <div style={{ fontSize: 15, fontWeight: 700, color: theme.navy, flex: 1 }}>{l.title}</div>
                      <div style={{ textAlign: "right" }}><span style={{ fontSize: 17, fontWeight: 800, color: theme.blue }}>₹{(l.rent / 1000).toFixed(0)}k</span><span style={{ fontSize: 10, color: theme.gray500 }}>/mo</span></div>
                    </div>
                    <div style={{ color: theme.gray500, fontSize: 12, marginTop: 3 }}>📍 {l.location}</div>
                    <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginTop: 10 }}>
                      <div style={{ display: "flex", gap: 6 }}>
                        {l.amenities.slice(0, 3).map(a => <span key={a} style={{ background: theme.lightBlue, borderRadius: 8, padding: "3px 8px", fontSize: 10, color: theme.blue }}>{amenityIcons[a]} {a}</span>)}
                      </div>
                      <div style={{ display: "flex", alignItems: "center", gap: 3 }}>
                        <span style={{ color: theme.accent, fontSize: 12 }}>⭐</span>
                        <span style={{ fontSize: 12, fontWeight: 700, color: theme.gray700 }}>{l.rating}</span>
                        {l.verified && <span style={{ color: theme.green, fontSize: 12, marginLeft: 4 }}>✓</span>}
                      </div>
                    </div>
                  </div>
                </div>
              ))}
            </div>
          </div>
        </div>
      )}

      {activeTab === "saved" && (
        <div style={{ flex: 1, overflowY: "auto", paddingBottom: 80 }}>
          <div style={{ padding: "52px 20px 20px" }}>
            <div style={{ fontSize: 22, fontWeight: 700, color: theme.navy, marginBottom: 4 }}>Saved Properties</div>
            <div style={{ fontSize: 13, color: theme.gray500, marginBottom: 20 }}>{savedIds.length} saved</div>
            {listings.filter(l => savedIds.includes(l.id)).map(l => (
              <div key={l.id} onClick={() => { setSelectedListing(l); setScreen("detail"); }} style={{ background: "white", borderRadius: 16, padding: 16, marginBottom: 12, display: "flex", gap: 14, cursor: "pointer", boxShadow: "0 2px 8px rgba(0,0,0,0.06)" }}>
                <div style={{ width: 72, height: 72, borderRadius: 12, background: theme.lightBlue, display: "flex", alignItems: "center", justifyContent: "center", fontSize: 32, flexShrink: 0 }}>{l.img}</div>
                <div style={{ flex: 1 }}>
                  <div style={{ fontSize: 14, fontWeight: 700, color: theme.navy }}>{l.title}</div>
                  <div style={{ fontSize: 12, color: theme.gray500, marginTop: 2 }}>📍 {l.location}</div>
                  <div style={{ display: "flex", justifyContent: "space-between", marginTop: 8 }}>
                    <span style={{ fontSize: 15, fontWeight: 800, color: theme.blue }}>₹{l.rent.toLocaleString()}/mo</span>
                    <span style={{ fontSize: 11, color: l.available ? theme.green : theme.red, fontWeight: 600 }}>{l.available ? "Available" : "Occupied"}</span>
                  </div>
                </div>
              </div>
            ))}
            {savedIds.length === 0 && <div style={{ textAlign: "center", padding: "60px 20px", color: theme.gray500 }}><div style={{ fontSize: 48, marginBottom: 12 }}>🤍</div><div style={{ fontSize: 16, fontWeight: 600 }}>No saved properties yet</div></div>}
          </div>
        </div>
      )}

      {activeTab === "profile" && (
        <div style={{ flex: 1, overflowY: "auto", paddingBottom: 80 }}>
          <div style={{ background: `linear-gradient(135deg, ${theme.navy}, #1A3A6B)`, padding: "52px 20px 32px", borderRadius: "0 0 28px 28px", display: "flex", flexDirection: "column", alignItems: "center" }}>
            <div style={{ width: 80, height: 80, borderRadius: "50%", background: "rgba(255,255,255,0.15)", display: "flex", alignItems: "center", justifyContent: "center", fontSize: 36, marginBottom: 12 }}>👤</div>
            <div style={{ color: "white", fontSize: 20, fontWeight: 700 }}>Rahul Sharma</div>
            <div style={{ color: "rgba(255,255,255,0.6)", fontSize: 13 }}>rahul.sharma@email.com</div>
            <div style={{ background: theme.accent, color: theme.navy, padding: "4px 16px", borderRadius: 20, fontSize: 12, fontWeight: 700, marginTop: 10 }}>Verified Tenant ✓</div>
          </div>
          <div style={{ padding: "20px" }}>
            {[{ icon: "📋", label: "My Bookings", count: "2 Active" }, { icon: "❤️", label: "Saved Properties", count: `${savedIds.length} Saved` }, { icon: "💳", label: "Payment History", count: "View All" }, { icon: "📄", label: "Rental Agreements", count: "2 Docs" }, { icon: "🔔", label: "Notifications" }, { icon: "🛡️", label: "Privacy & Security" }, { icon: "❓", label: "Help & Support" }, { icon: "🚪", label: "Logout", count: "" }].map((item, i) => (
              <div key={i} onClick={() => item.label === "Logout" && setScreen("login")} style={{ background: "white", borderRadius: 14, padding: "16px", marginBottom: 8, display: "flex", alignItems: "center", gap: 14, cursor: "pointer", boxShadow: "0 1px 4px rgba(0,0,0,0.05)" }}>
                <span style={{ fontSize: 20 }}>{item.icon}</span>
                <span style={{ flex: 1, fontSize: 14, fontWeight: 600, color: item.label === "Logout" ? theme.red : theme.navy }}>{item.label}</span>
                <span style={{ fontSize: 12, color: theme.gray500 }}>{item.count}</span>
                <span style={{ color: theme.gray300, fontSize: 16 }}>›</span>
              </div>
            ))}
          </div>
        </div>
      )}

      {/* Bottom Nav */}
      <div style={{ position: "absolute", bottom: 0, left: 0, right: 0, background: "white", borderTop: `1px solid ${theme.gray100}`, display: "flex", padding: "10px 0 24px", borderRadius: "0 0 40px 40px" }}>
        {[{ id: "home", icon: "🏠", label: "Home" }, { id: "search", icon: "🔍", label: "Search" }, { id: "saved", icon: "❤️", label: "Saved" }, { id: "profile", icon: "👤", label: "Profile" }].map(tab => (
          <button key={tab.id} onClick={() => setActiveTab(tab.id)} style={{ flex: 1, background: "none", border: "none", cursor: "pointer", display: "flex", flexDirection: "column", alignItems: "center", gap: 3 }}>
            <div style={{ fontSize: 22, filter: activeTab === tab.id ? "none" : "grayscale(100%) opacity(0.5)" }}>{tab.icon}</div>
            <div style={{ fontSize: 10, fontWeight: 600, color: activeTab === tab.id ? theme.blue : theme.gray500 }}>{tab.label}</div>
            {activeTab === tab.id && <div style={{ width: 4, height: 4, borderRadius: "50%", background: theme.blue }} />}
          </button>
        ))}
      </div>
    </div>
  );
}
